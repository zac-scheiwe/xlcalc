Private Sub Workbook_BeforeClose(Cancel As Boolean)
    On Error GoTo EH
    Cancel = EvLog.CancelClose
    ThisWorkbook.Saved = True
    EvLog.Enter "Book", "BeforeClose", Nothing, Nothing, "Cancel=" & Cancel
    EvLog.Leave
    Exit Sub
EH:
    EvLog.Fail "Book.BeforeClose", Err.Number
End Sub

Public Function WbFn(ByVal a As Variant) As Variant
    On Error GoTo EH
    WbFn = "wb:" & a
    Exit Function
EH:
    WbFn = "err " & Err.Number
End Function
